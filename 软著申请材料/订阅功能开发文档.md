## 1. 系统概述

### 1.1 功能特性

本订阅系统是一套完整的SaaS订阅管理解决方案，包含以下核心功能：

✅ **多层级订阅管理**
- 免费版、专业版、企业版
- 月付/年付周期
- 自动续费机制
- 订阅升级/降级

✅ **灵活的加油包系统**
- 次卡购买（100/300/1000次）
- FIFO优先级算法
- 自动过期处理
- 优先于订阅配额使用

✅ **完整的支付集成**
- 微信支付（Native扫码）
- 支付宝（电脑网站支付）
- 防重复支付机制
- 支付回调处理

✅ **智能使用量追踪**
- 实时配额监控
- 自动超量限制
- 可视化进度展示
- 多维度统计

```typescript
/**
 * 获取用户订阅计划（CloudBase）
 */
export async function getUserPlanCloudBase(userId: string): Promise<SubscriptionTier> {
  const db = getCloudBaseDatabase();

  // 1. 查询用户信息
  const userResult = await db
    .collection('users')
    .doc(userId)
    .get();

  if (!userResult.data) {
    return 'free';
  }

  const user = userResult.data[0];
  let currentPlan = (user.subscription_plan || 'free').toLowerCase() as SubscriptionTier;

  // 2. 验证订阅是否过期
  const now = new Date().toISOString();

  const subscriptionsResult = await db
    .collection('user_subscriptions')
    .where({
      user_id: userId,
      status: 'active',
    })
    .get();

  // 3. 检查是否有活跃订阅
  let hasActiveSubscription = false;
  for (const subscription of subscriptionsResult.data) {
    if (subscription.subscription_end > now) {
      hasActiveSubscription = true;
      currentPlan = subscription.plan_type.toLowerCase() as SubscriptionTier;
      break;
    } else {
      // 订阅已过期，更新状态
      await updateExpiredSubscription(subscription);
    }
  }

  // 4. 如果没有活跃订阅，降级为免费版
  if (!hasActiveSubscription && currentPlan !== 'free') {
    await db.collection('users').doc(userId).update({
      subscription_plan: 'free',
    });
    currentPlan = 'free';
  }

  return currentPlan;
}
```

```typescript
/**
 * 更新过期订阅状态
 */
async function updateExpiredSubscription(subscription: any) {
  const db = getCloudBaseDatabase();
  const now = new Date().toISOString();

  // 更新订阅状态
  await db.collection('user_subscriptions')
    .doc(subscription._id)
    .update({
      status: 'expired',
      updated_at: now,
    });

  // 检查用户是否有其他活跃订阅
  const activeSubsResult = await db
    .collection('user_subscriptions')
    .where({
      user_id: subscription.user_id,
      status: 'active',
    })
    .get();

  const hasOtherActiveSubscription = activeSubsResult.data.some(
    (sub) => sub.subscription_end > now
  );

  // 只有在没有其他活跃订阅时才降级用户
  if (!hasOtherActiveSubscription) {
    await db.collection('users').doc(subscription.user_id).update({
      subscription_plan: 'free',
    });
  }
}
```

```typescript
/**
 * 处理订阅购买
 */
async function handleSubscriptionPurchase(payment: any, now: string) {
  const db = getCloudBaseDatabase();
  const { planType, billingCycle } = payment.metadata || {};

  // 计算订阅结束时间
  const startDate = new Date(now);
  const endDate = new Date(startDate);

  if (billingCycle === 'yearly') {
    endDate.setFullYear(endDate.getFullYear() + 1);
  } else {
    endDate.setMonth(endDate.getMonth() + 1);
  }

  // 创建订阅记录
  await db.collection('user_subscriptions').add({
    user_id: payment.user_id,
    plan_type: planType || 'pro',
    billing_cycle: billingCycle || 'monthly',
    status: 'active',
    subscription_start: startDate.toISOString(),
    subscription_end: endDate.toISOString(),
    auto_renew: true,
    created_at: now,
    updated_at: now,
  });

  // 更新用户订阅等级
  await db.collection('users').doc(payment.user_id).update({
    subscription_plan: planType || 'pro',
    subscription_status: 'active',
    updated_at: now,
  });
}
```

```typescript
class WeChatPayAdapter implements PaymentAdapter {
  async createOrder(amount: number, description: string) {

        trade_type: 'NATIVE',

      }),
    });

    return {
      orderId: result.out_trade_no,
      qrCodeUrl: result.code_url,
    };
  }

    // 验证签名
```

### 5.2 防重复支付机制

```typescript
/**
 * 检查是否有重复的支付请求
 */
async function checkDuplicatePayment(
  userId: string,
  amount: number,
  timeWindowSeconds: number = 60
): Promise<{ isDuplicate: boolean; waitTime?: number }> {
  const db = getCloudBaseDatabase();
  const now = new Date();
  const timeWindow = new Date(now.getTime() - timeWindowSeconds * 1000);

  // 查询时间窗口内的支付记录
  const duplicateResult = await db
    .collection('payments')
    .where({
      user_id: userId,
      amount: amount,
      status: 'pending',
      created_at: db.command.gte(timeWindow.toISOString()),
    })
    .get();

  if (duplicateResult.data.length > 0) {
    const lastPayment = duplicateResult.data[0];
    const lastPaymentTime = new Date(lastPayment.created_at);
    const waitTime = timeWindowSeconds - Math.floor((now.getTime() - lastPaymentTime.getTime()) / 1000);

    return {
      isDuplicate: true,
      waitTime: Math.max(waitTime, 0),
    };
  }

  return { isDuplicate: false };
}
```

```typescript
/**
 * 微信支付回调处理
 */
export async function POST(request: NextRequest) {
  const db = getCloudBaseDatabase();

  try {
    // 1. 解析XML通知
    const body = await request.text();
    const notification = await parseXML(body);

    // 2. 验证签名
    const isValid = await wechatPayAdapter.verifyNotification(notification);
    if (!isValid) {
      return new Response('签名验证失败', { status: 400 });
    }

    // 3. 查询订单
    const payment = await db
      .collection('payments')
      .where({ order_id: notification.out_trade_no })
      .get();

    if (payment.data.length === 0) {
      return new Response('订单不存在', { status: 404 });
    }

    // 4. 验证金额
    if (payment.data[0].amount * 100 !== parseInt(notification.total_fee)) {
      return new Response('金额不符', { status: 400 });
    }

    // 5. 更新支付状态
    const now = new Date().toISOString();
    await db.collection('payments').doc(payment.data[0]._id).update({
      status: 'completed',
      transaction_id: notification.transaction_id,
      updated_at: now,
    });

    // 6. 根据类型创建订阅或加油包
    const paymentType = payment.data[0].metadata?.type;
    if (paymentType === 'credit_package') {
      await handleCreditPackagePurchase(payment.data[0], now);
    } else {
      await handleSubscriptionPurchase(payment.data[0], now);
    }

    // 7. 返回成功响应
    return new Response(buildXML({ return_code: 'SUCCESS', return_msg: 'OK' }), {
      headers: { 'Content-Type': 'application/xml' },
    });
  } catch (error) {
    console.error('WeChat pay notification error:', error);
    return new Response(buildXML({ return_code: 'FAIL', return_msg: error.message }), {
      headers: { 'Content-Type': 'application/xml' },
    });
  }
}
```

### 6.3 FIFO优先级算法

```typescript
/**
 * 记录使用量（优先扣除加油包）
 */
export async function recordRecommendationUsageCloudBase(
  userId: string,
  usageType: 'generation' | 'modification' = 'generation'
): Promise<{ success: boolean; error?: string }> {
  const db = getCloudBaseDatabase();
  const nowISO = new Date().toISOString();
  const now = new Date();

  // 1. 查询活跃的加油包，按购买时间排序（FIFO）
  const creditPackagesResult = await db
    .collection('user_credit_packages')
    .where({
      user_id: userId,
      status: 'active',
    })
    .orderBy('purchase_date', 'asc')
    .get();

  // 2. 查找最早购买的、有剩余次数的、未过期的加油包
  let targetPackage = null;
  for (const pkg of creditPackagesResult.data) {
    // 检查是否过期
    if (pkg.expiry_date < nowISO) {
      await db.collection('user_credit_packages').doc(pkg._id).update({
        status: 'expired',
        updated_at: nowISO,
      });
      continue;
    }

    // 找到有剩余次数的加油包
    if (pkg.credits_remaining > 0) {
      targetPackage = pkg;
      break; // FIFO: 第一个有次数的包
    }
  }

  // 3. 如果有加油包，优先扣除
  if (targetPackage) {
    const newCreditsRemaining = Math.max(0, targetPackage.credits_remaining - 1);

    await db.collection('user_credit_packages').doc(targetPackage._id).update({
      credits_remaining: newCreditsRemaining,
      updated_at: nowISO,
    });

    // 如果用完了，标记为used_up
    if (newCreditsRemaining === 0) {
      await db.collection('user_credit_packages').doc(targetPackage._id).update({
        status: 'used_up',
      });
    }

    return { success: true };
  }

  // 4. 没有加油包，记录到订阅配额
  await db.collection('recommendation_usage').add({
    user_id: userId,
    usage_date: nowISO,
    usage_type: usageType,
    created_at: nowISO,
  });

  return { success: true };
}
```

## 7. 使用量追踪

### 7.1 获取使用统计

```typescript
/**
 * 获取用户使用统计（CloudBase）
 */
export async function getUserUsageStatsCloudBase(userId: string) {
  const db = getCloudBaseDatabase();
  const now = new Date();
  const currentMonth = new Date(now.getFullYear(), now.getMonth(), 1);
  const nextMonth = new Date(now.getFullYear(), now.getMonth() + 1, 1);
  const nowISO = now.toISOString();

  // 1. 获取用户订阅计划
  const plan = await getUserPlanCloudBase(userId);
  const planConfig = SUBSCRIPTION_LIMITS[plan];

  // 2. 统计本月使用次数（仅订阅配额）
  const usageResult = await db
    .collection('recommendation_usage')
    .where({
      user_id: userId,
      usage_date: db.command.gte(currentMonth.toISOString()).and(db.command.lt(nextMonth.toISOString())),
    })
    .get();

  const currentUsage = usageResult.data.length;

  // 3. 查询活跃的加油包
  const creditPackagesResult = await db
    .collection('user_credit_packages')
    .where({
      user_id: userId,
      status: 'active',
    })
    .get();

  // 4. 计算加油包剩余次数
  let totalCreditPackageRemaining = 0;
  for (const pkg of creditPackagesResult.data) {
    if (pkg.expiry_date < nowISO) {
      // 过期，更新状态
      await db.collection('user_credit_packages').doc(pkg._id).update({
        status: 'expired',
      });
      continue;
    }
    totalCreditPackageRemaining += pkg.credits_remaining;
  }

  // 5. 计算总配额
  let periodLimit = planConfig.monthlyGenerations;
  const isUnlimited = periodLimit === -1;

  if (!isUnlimited && totalCreditPackageRemaining > 0) {
    periodLimit += totalCreditPackageRemaining;
  }

  // 6. 计算剩余次数
  const remaining = isUnlimited ? -1 : Math.max(0, periodLimit - currentUsage);

  return {
    current: currentUsage,
    limit: periodLimit,
    remaining: remaining,
    isUnlimited: isUnlimited,
    plan: plan,
    creditPackageRemaining: totalCreditPackageRemaining,
  };
}
```

### 7.2 检查是否可以使用

```typescript
/**
 * 检查用户是否可以使用功能
 */
export async function checkUsageLimit(userId: string): Promise<{
  canUse: boolean;
  reason?: string;
  remaining?: number;
}> {
  const stats = await getUserUsageStatsCloudBase(userId);

  // 企业版无限
  if (stats.isUnlimited) {
    return { canUse: true };
  }

  // 检查剩余次数
  if (stats.remaining <= 0) {
    return {
      canUse: false,
      reason: 'USAGE_LIMIT_EXCEEDED',
      remaining: 0,
    };
  }

  return {
    canUse: true,
    remaining: stats.remaining,
  };
}
```

### 8.1 订阅状态Hook

```typescript
import { useState, useEffect } from 'react';
import { getUserUsageStats } from '@/lib/subscription/usage-tracker';

export interface UsageStats {
  current: number;
  limit: number;
  remaining: number;
  isUnlimited: boolean;
  plan: 'free' | 'pro' | 'enterprise';
}

export function useSubscription(userId?: string) {
  const [stats, setStats] = useState<UsageStats | null>(null);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    async function fetchStats() {
      if (!userId) {
        setLoading(false);
        return;
      }

      try {
        const data = await getUserUsageStats(userId);
        setStats(data);
      } catch (error) {
        console.error('Failed to fetch usage stats:', error);
      } finally {
        setLoading(false);
      }
    }

    fetchStats();
  }, [userId]);

  return { stats, loading };
}
```

### C. 常见问题

**Q: 如何支持多个支付方式？**
A: 实现多个PaymentAdapter，根据用户选择调用对应的适配器。

**Q: 如何实现免费试用？**
A: 在创建订阅时设置 `trial_end_date`，在到期前不扣费。

**Q: 如何处理退款？**
A: 监听退款回调，更新订单状态，并相应调整订阅状态。

**Q: 如何实现企业版无限配额？**
A: 在检查配额时判断 `plan === 'enterprise'` 直接返回 `isUnlimited: true`。
