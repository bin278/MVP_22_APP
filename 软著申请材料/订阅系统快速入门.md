# 订阅系统快速入门指南

> 5分钟快速集成订阅功能到你的项目

---

## 📦 前置条件

- Node.js 18+ 项目
- React/Next.js 前端
- 任意数据库（支持SQL或NoSQL）
- 微信支付或支付宝商户号

## 🚀 快速开始（3步集成）

### 第1步：复制核心文件（2分钟）

```bash
# 1. 复制订阅管理模块
cp -r lib/subscription your-project/lib/

# 2. 复制支付系统模块
cp -r lib/payment your-project/lib/

# 3. 复制数据库适配器
cp -r lib/database your-project/lib/
```

### 第2步：配置环境变量（2分钟）

创建 `.env.local`：

```bash
# 数据库（选择你使用的）
DATABASE_URL=your-database-url
# 或者
CLOUDBASE_ENV_ID=your-cloudbase-env-id

# 支付配置（二选一或都配置）
WECHAT_APP_ID=your-wechat-appid
WECHAT_MCH_ID=your-wechat-mchid
WECHAT_API_KEY=your-wechat-key

ALIPAY_APP_ID=your-alipay-appid
ALIPAY_PRIVATE_KEY=your-alipay-private-key
```

### 第3步：使用API（1分钟）

```typescript
// 在你的代码中使用
import { getUserUsageStats, checkUsageLimit } from '@/lib/subscription/usage-tracker';

// 检查用户是否可以使用
async function handleClick() {
  const check = await checkUsageLimit(userId);

  if (!check.canUse) {
    alert('配额已用完，请升级订阅');
    router.push('/payment');
    return;
  }

  // 执行你的业务逻辑
  await doYourAction();
}

// 获取使用统计
const stats = await getUserUsageStats(userId);
console.log(`剩余次数：${stats.remaining}/${stats.limit}`);
```

### 3. 添加新的支付方式

```typescript
// lib/payment/stripe-adapter.ts
export class StripeAdapter implements PaymentAdapter {
  async createOrder(amount: number, description: string) {
    // 实现Stripe支付逻辑
    const stripe = require('stripe')(process.env.STRIPE_SECRET_KEY);
    const paymentIntent = await stripe.paymentIntents.create({
      amount: amount * 100,
      currency: 'usd',
      description,
    });

    return {
      orderId: paymentIntent.id,
      clientSecret: paymentIntent.client_secret,
    };
  }

  async verifyNotification(notification: any) {
    // 验证Stripe webhook签名
    const stripe = require('stripe')(process.env.STRIPE_SECRET_KEY);
    return stripe.webhooks.constructEvent(
      JSON.stringify(notification),
      notification.headers['stripe-signature'],
      process.env.STRIPE_WEBHOOK_SECRET
    );
  }
}
```

### 4. 替换数据库

```typescript
// lib/database/adapters/supabase.ts
export async function getUserPlanSupabase(userId: string) {
  const supabase = createClient(
    process.env.SUPABASE_URL,
    process.env.SUPABASE_ANON_KEY
  );

  const { data, error } = await supabase
    .from('user_subscriptions')
    .select('*')
    .eq('user_id', userId)
    .eq('status', 'active')
    .order('subscription_end', { ascending: false })
    .limit(1);

  if (error || !data || data.length === 0) {
    return 'free';
  }

  return data[0].plan_type;
}
```

```typescript
// 在你的组件中
function UsageAlert({ stats }: { stats: UsageStats }) {
  if (stats.isUnlimited) return null;

  const remainingPercentage = (stats.remaining / stats.limit) * 100;

  if (remainingPercentage <= 10) {
    return (
      <Alert variant="destructive">
        <AlertCircle className="h-4 w-4" />
        <AlertTitle>配额即将用完</AlertTitle>
        <AlertDescription>

          <Button size="sm" className="ml-2">

          </Button>
        </AlertDescription>
      </Alert>
    );
  }

  if (remainingPercentage <= 20) {
    return (
      <Alert>
        <Info className="h-4 w-4" />
        <AlertTitle>配额提醒</AlertTitle>
        <AlertDescription>
          已使用 {stats.current}/{stats.limit} 次
        </AlertDescription>
      </Alert>
    );
  }

  return null;
}
```

## 📊 数据库表结构SQL

### MySQL版本

```sql
-- 用户表
CREATE TABLE users (
  id VARCHAR(36) PRIMARY KEY,
  email VARCHAR(255) UNIQUE NOT NULL,
  subscription_plan ENUM('free', 'pro', 'enterprise') DEFAULT 'free',
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
);

-- 订阅表
CREATE TABLE user_subscriptions (
  id VARCHAR(36) PRIMARY KEY,
  user_id VARCHAR(36) NOT NULL,
  plan_type ENUM('free', 'pro', 'enterprise') NOT NULL,
  billing_cycle ENUM('monthly', 'yearly') NOT NULL,
  status ENUM('active', 'inactive', 'cancelled', 'expired') NOT NULL,
  subscription_start TIMESTAMP NOT NULL,
  subscription_end TIMESTAMP NOT NULL,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  FOREIGN KEY (user_id) REFERENCES users(id)
);

-- 加油包表
CREATE TABLE user_credit_packages (
  id VARCHAR(36) PRIMARY KEY,
  user_id VARCHAR(36) NOT NULL,
  package_type ENUM('basic', 'standard', 'premium') NOT NULL,
  credits_total INT NOT NULL,
  credits_remaining INT NOT NULL,
  status ENUM('active', 'expired', 'used_up') NOT NULL,
  expiry_date TIMESTAMP NOT NULL,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  FOREIGN KEY (user_id) REFERENCES users(id)
);

-- 使用记录表
CREATE TABLE recommendation_usage (
  id VARCHAR(36) PRIMARY KEY,
  user_id VARCHAR(36) NOT NULL,
  usage_date TIMESTAMP NOT NULL,
  usage_type ENUM('generation', 'modification') NOT NULL,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (user_id) REFERENCES users(id)
);

-- 索引
CREATE INDEX idx_subscriptions_user_status ON user_subscriptions(user_id, status);
CREATE INDEX idx_credit_packages_user_status ON user_credit_packages(user_id, status);
CREATE INDEX idx_usage_user_date ON recommendation_usage(user_id, usage_date);
```

### MongoDB版本

```javascript
// 用户集合
db.createCollection('users');
db.users.createIndex({ email: 1 }, { unique: true });

// 订阅集合
db.createCollection('user_subscriptions');
db.user_subscriptions.createIndex({ user_id: 1, status: 1 });
db.user_subscriptions.createIndex({ subscription_end: 1 });

// 加油包集合
db.createCollection('user_credit_packages');
db.user_credit_packages.createIndex({ user_id: 1, status: 1 });
db.user_credit_packages.createIndex({ expiry_date: 1 });

// 使用记录集合
db.createCollection('recommendation_usage');
db.recommendation_usage.createIndex({ user_id: 1, usage_date: 1 });
```

## ⚠️ 注意事项

### 安全性

1. **验证所有支付回调签名**
2. **不要在前端暴露API密钥**
3. **使用HTTPS加密传输**
4. **实现API速率限制**

1. **为数据库添加合适的索引**
